/* ============================================================
   1. DATE TABLE
   Create this as: Modeling → New Table
   ============================================================ */

Dim_Date_PBI =
ADDCOLUMNS(
    CALENDAR(DATE(2016, 1, 1), DATE(2018, 12, 31)),
    "date_key", VALUE(FORMAT([Date], "YYYYMMDD")),
    "Year", YEAR([Date]),
    "Month Number", MONTH([Date]),
    "Month Name", FORMAT([Date], "MMMM"),
    "Year Month", FORMAT([Date], "YYYY-MM"),
    "Quarter", "Q" & FORMAT([Date], "Q"),
    "Day", DAY([Date]),
    "Day Name", FORMAT([Date], "dddd"),
    "Week Number", WEEKNUM([Date]),
    "Is Weekend", IF(WEEKDAY([Date], 2) > 5, 1, 0)
)

/* ============================================================
   5. DELIVERY PERFORMANCE MEASURES
   Used in Delivery, Payment & Review Analysis Dashboard
   ============================================================ */

Delivered Orders =
VAR OrdersInContext =
    VALUES(fact_order_items[order_id])
RETURN
    CALCULATE(
        SUM(fact_order_summary[is_delivered]),
        TREATAS(OrdersInContext, fact_order_summary[order_id])
    )


Late Delivery % =
VAR OrdersInContext =
    VALUES(fact_order_items[order_id])
RETURN
    CALCULATE(
        DIVIDE(
            SUM(fact_order_summary[is_late_delivery]),
            SUM(fact_order_summary[is_delivered])
        ),
        TREATAS(OrdersInContext, fact_order_summary[order_id])
    )


On Time Delivery % =
VAR OrdersInContext =
    VALUES(fact_order_items[order_id])
VAR DeliveredOrders =
    CALCULATE(
        SUM(fact_order_summary[is_delivered]),
        TREATAS(OrdersInContext, fact_order_summary[order_id])
    )
VAR LateOrders =
    CALCULATE(
        SUM(fact_order_summary[is_late_delivery]),
        TREATAS(OrdersInContext, fact_order_summary[order_id])
    )
VAR OnTimeOrders =
    DeliveredOrders - LateOrders
RETURN
    DIVIDE(OnTimeOrders, DeliveredOrders)


Avg Delivery Days =
VAR OrdersInContext =
    VALUES(fact_order_items[order_id])
RETURN
    CALCULATE(
        AVERAGE(fact_order_summary[delivery_days]),
        TREATAS(OrdersInContext, fact_order_summary[order_id])
    )
